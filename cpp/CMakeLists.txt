cmake_minimum_required(VERSION 3.18)
project(MasteryProgrammingLabs VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MASTERY_STUDENT "Test the student TODO headers instead of reference headers" OFF)
option(MASTERY_SANITIZE "Enable address/undefined sanitizers (GCC/Clang non-Windows)" OFF)
enable_testing()
find_package(Threads REQUIRED)
foreach(lab IN ITEMS two_sum spsc_ring order_book)
 add_executable(test_${lab} tests/test_${lab}.cpp)
 target_link_libraries(test_${lab} PRIVATE Threads::Threads)
 if(MSVC)
  target_compile_options(test_${lab} PRIVATE /W4)
 else()
  target_compile_options(test_${lab} PRIVATE -Wall -Wextra -Wpedantic)
 endif()
 if(MASTERY_STUDENT)
  target_compile_definitions(test_${lab} PRIVATE MASTERY_STUDENT=1)
 endif()
 add_test(NAME ${lab} COMMAND test_${lab})
 set_tests_properties(${lab} PROPERTIES TIMEOUT 20)
endforeach()
add_executable(mastery_benchmark bench/bench.cpp)
if(MASTERY_SANITIZE AND NOT MSVC AND NOT WIN32)
 foreach(target IN ITEMS test_two_sum test_spsc_ring test_order_book)
  target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
  target_link_options(${target} PRIVATE -fsanitize=address,undefined)
 endforeach()
endif()
