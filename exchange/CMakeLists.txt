cmake_minimum_required(VERSION 3.20)
project(MasteryExchange VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MASTERY_ASAN "Build with AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
add_library(mastery_exchange INTERFACE)
target_include_directories(mastery_exchange INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
 target_compile_options(mastery_exchange INTERFACE /W4)
else()
 target_compile_options(mastery_exchange INTERFACE -Wall -Wextra -Wpedantic -Wconversion)
endif()
if(MASTERY_ASAN AND NOT MSVC)
 target_compile_options(mastery_exchange INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
 target_link_options(mastery_exchange INTERFACE -fsanitize=address,undefined)
endif()
add_executable(mastery_replay src/replay.cpp)
target_link_libraries(mastery_replay PRIVATE mastery_exchange)
add_executable(test_exchange tests/test_exchange.cpp)
target_link_libraries(test_exchange PRIVATE mastery_exchange)
enable_testing()
add_test(NAME exchange_invariants COMMAND test_exchange)
add_test(NAME csv_replay COMMAND mastery_replay ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/session.csv)
set_tests_properties(exchange_invariants csv_replay PROPERTIES TIMEOUT 20)
add_executable(mastery_exchange_bench src/bench.cpp)
target_link_libraries(mastery_exchange_bench PRIVATE mastery_exchange)
add_executable(mastery_allocator_experiment src/allocator_experiment.cpp)
target_link_libraries(mastery_allocator_experiment PRIVATE mastery_exchange)
add_test(NAME pmr_experiment COMMAND mastery_allocator_experiment)
set_tests_properties(pmr_experiment PROPERTIES TIMEOUT 20)
