cmake_minimum_required(VERSION 3.18)
project(MasteryPhaseVII LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MASTERY_SANITIZE "Enable ASan and UBSan" OFF)
enable_testing()
add_executable(test_segmented tests/test_segmented.cpp)
target_include_directories(test_segmented PRIVATE include)
target_compile_options(test_segmented PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(MASTERY_SANITIZE)
  target_compile_options(test_segmented PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
  target_link_options(test_segmented PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME segmented_recovery_and_idempotency COMMAND test_segmented)
set_tests_properties(segmented_recovery_and_idempotency PROPERTIES TIMEOUT 30)
