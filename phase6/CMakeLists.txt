cmake_minimum_required(VERSION 3.18)
project(MasteryPhase6DurableRecovery LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MASTERY_SANITIZE "Compile with AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
enable_testing()
add_executable(test_durable tests/test_durable.cpp)
target_include_directories(test_durable PRIVATE include)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
 message(FATAL_ERROR "The Phase VI WAL lesson is currently implemented for Linux/POSIX only")
endif()
target_compile_options(test_durable PRIVATE -Wall -Wextra -Wpedantic)
if(MASTERY_SANITIZE)
 target_compile_options(test_durable PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
 target_link_options(test_durable PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME durable_wal_idempotency_fanout COMMAND test_durable)
set_tests_properties(durable_wal_idempotency_fanout PROPERTIES TIMEOUT 45)
add_executable(durable_demo src/demo.cpp)
target_include_directories(durable_demo PRIVATE include)
target_compile_options(durable_demo PRIVATE -Wall -Wextra -Wpedantic)
if(MASTERY_SANITIZE)
 target_compile_options(durable_demo PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
 target_link_options(durable_demo PRIVATE -fsanitize=address,undefined)
endif()
