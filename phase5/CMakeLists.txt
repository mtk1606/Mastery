cmake_minimum_required(VERSION 3.18)
project(MasteryReliabilityLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MASTERY_SANITIZE "Build AddressSanitizer+UndefinedBehaviorSanitizer tests" OFF)
enable_testing()
add_executable(test_reliability tests/test_reliability.cpp)
target_include_directories(test_reliability PRIVATE include)
if(MSVC)
 target_compile_options(test_reliability PRIVATE /W4)
else()
 target_compile_options(test_reliability PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(MASTERY_SANITIZE AND NOT MSVC AND NOT WIN32)
 target_compile_options(test_reliability PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
 target_link_options(test_reliability PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME reliability COMMAND test_reliability)
set_tests_properties(reliability PROPERTIES TIMEOUT 30)

add_executable(reliability_demo src/demo.cpp)
target_include_directories(reliability_demo PRIVATE include)
if(MASTERY_SANITIZE AND NOT MSVC AND NOT WIN32)
 target_compile_options(reliability_demo PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
 target_link_options(reliability_demo PRIVATE -fsanitize=address,undefined)
endif()
